### DEG analysis using DESeq2 ###

library(DESeq2)
setwd()
fcc<-read.table("Input_DESeq2.txt",header=TRUE,sep="\t",row.names=1)
ncol(fcc)
nrow(fcc)
dim(fcc)
head(fcc)
SI<-read.table("SI.txt",header=TRUE,sep="\t")
ncol(SI)
nrow(SI)
dim(SI)
head(SI)
SI

dds = DESeqDataSetFromMatrix(countData = fcc, colData = SI, design = ~ Type)

dds = estimateSizeFactors(dds)
sizeFactors(dds)

par(mar=c(10,6,4,1))
barplot(colSums(counts(dds)),las=2,cex.axis=1.5,cex.names=1.5,main="Sum of the Counts")
par(mar=c(10,6,4,1))
barplot(sizeFactors(dds),las=2,cex.names=1.5,cex.axis=1.5,main="Size Factors")

par(mar=c(10,6,4,1))
boxplot(log2(counts(dds)+1), ylim=c(-5,20),las=2,cex.names=1.5, main="Non-normalized read counts", ylab="log2(read counts)")
par(mar=c(10,6,4,1))
boxplot(log2(counts(dds,normalized=TRUE)+1), ylim=c(-5,20),las=2,cex.names=1.5,main="Size Factor Normalized read counts", ylab="log2(read counts)")

dds = estimateDispersions(dds)
plotDispEsts(dds)

dds = nbinomWaldTest(dds)
res_TS_NTC = results(dds,contrast=c("Type","TS","NTC"))
res_TScc_NTCcc = results(dds,contrast=c("Type","TS_CC","NTC_CC"))
res_TScc_TS = results(dds,contrast=c("Type","TS_CC","TS"))

mcols(res_TS_NTC,use.names=TRUE)
mcols(res_TScc_NTCcc,use.names=TRUE)
mcols(res_TScc_TS,use.names=TRUE)

plotMA(res_TS_NTC)
plotMA(res_TScc_NTCcc)
plotMA(res_TScc_TS)

hist(res_TS_NTC$pvalue,breaks=20,col="grey")
hist(res_TScc_NTCcc$pvalue,breaks=20,col="grey")
hist(res_TScc_TS$pvalue,breaks=20,col="grey")

vsd <- varianceStabilizingTransformation(dds, blind=TRUE)
p = plotPCA(vsd,intgroup="Type")
p

sampleDists <- dist(t(assay(vsd)))
library(pheatmap)
library(RColorBrewer)
sampleDistMatrix <- as.matrix(sampleDists)
rownames(sampleDistMatrix) <- paste(vsd$Type, sep="-")
colnames(sampleDistMatrix) <- NULL
colors <- colorRampPalette( rev(brewer.pal(9, "Blues")) )(255)
pheatmap(sampleDistMatrix,
         clustering_distance_rows=sampleDists,
         clustering_distance_cols=sampleDists,
         col=colors)

res_TS_NTC = res_TS_NTC[order(res_TS_NTC$padj),]
write.table(res_TS_NTC, "res_TS_NTC.txt", row.name=T, quote=F, sep='\t')

res_TScc_NTCcc = res_TScc_NTCcc[order(res_TScc_NTCcc$padj),]
write.table(res_TScc_NTCcc, "res_TScc_NTCcc.txt", row.name=T, quote=F, sep='\t')

res_TScc_TS = res_TScc_TS[order(res_TScc_TS$padj),]
write.table(res_TScc_TS, "res_TScc_TS.txt", row.name=T, quote=F, sep='\t')
