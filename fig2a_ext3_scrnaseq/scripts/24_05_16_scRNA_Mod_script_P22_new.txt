# Categorize the cancer cells based on TYMS and TK1 expression
gene_expression <- FetchData(P22_cancer, vars = c("TYMS", "TK1"))

P22_cancer$category <- sapply(1:nrow(gene_expression), function(i) {
  TYMS <- gene_expression[i, "TYMS"]
  TK1 <- gene_expression[i, "TK1"]
  if (TYMS == 0 & TK1 == 0) {
    return("DN")
  } else if (TYMS > 0 & TK1 > 0) {
    return("DP")
  } else if (TYMS > 0 & TK1 == 0) {
    return("TYMS")
  } else {
    return("TK1")
  }
})
