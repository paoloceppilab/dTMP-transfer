gene_expression <- FetchData(Pooled_cancer, vars = c("TYMS", "TK1","MKI67"))
Pooled_cancer$category <- sapply(1:nrow(gene_expression), function(i) {
  TYMS <- gene_expression[i, "TYMS"]
  TK1 <- gene_expression[i, "TK1"]
  MKI67 <- gene_expression[i, "MKI67"]
    
  if (TYMS == 0 & TK1 == 0 & MKI67 == 0) {
    return("Non_Prol_DN")
  } else if (TYMS == 0 & TK1 == 0 & MKI67 > 0) {
    return("Prol_DN")
  } else {
    return("All")
  }
})

# Find markers for every cluster compared to all remaining cells, report only the positive ones
markers_myeloid <- FindAllMarkers(object = Pooled_myeloid, 
                          only.pos = TRUE,
                          logfc.threshold = 0.25,min.diff.pct=0.15)   

write.table(markers_myeloid,file="FAMs_myeloid.txt",sep="\t")

Macrophage_mark<-markers_myeloid %>% mutate(cell_type = str_replace(gene, '^CD68$', "Macrophage")) %>% filter(str_detect(gene,'^CD68$'))
M1_Macrophage_mark<-markers_myeloid %>% mutate(cell_type = str_replace(gene, '^NOS2$|^HLA-DRA$', "M1_Macrophage")) %>% filter(str_detect(gene,'^NOS2$|^HLA-DRA$'))
M2_Macrophage_mark<-markers_myeloid %>% mutate(cell_type = str_replace(gene, '^CD163$|^MRC1$|^MSR1$', "M2_Macrophage")) %>% filter(str_detect(gene,'^CD163$|^MRC1$|^MSR1$'))

Myeloid_Identity<-rbind(Macrophage_mark,M1_Macrophage_mark,M2_Macrophage_mark)

write.table(Myeloid_Identity, file="MTMs.txt",sep="\t")
