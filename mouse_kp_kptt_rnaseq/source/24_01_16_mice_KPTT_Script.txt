### DEG analysis using DESeq2 ###

library(DESeq2)
setwd()
fcc<-read.table("Input_DESeq2.txt",header=TRUE,sep="\t",row.names=1)
ncol(fcc)
nrow(fcc)
dim(fcc)
head(fcc)
SI<-read.table("SI.txt",header=TRUE,sep="\t")
ncol(SI)
nrow(SI)
dim(SI)
head(SI)
SI

dds = DESeqDataSetFromMatrix(countData = fcc, colData = SI, design = ~ Type)
dds = estimateSizeFactors(dds)
sizeFactors(dds)

par(mar=c(10,6,4,1))
barplot(colSums(counts(dds)),las=2,cex.axis=1.5,cex.names=1.5,main="Sum of the Counts")
par(mar=c(10,6,4,1))
barplot(sizeFactors(dds),las=2,cex.names=1.5,cex.axis=1.5,main="Size Factors")

par(mar=c(10,6,4,1))
boxplot(log2(counts(dds)+1), ylim=c(-5,20),las=2,cex.names=1.5, main="Non-normalized read counts", ylab="log2(read counts)")
par(mar=c(10,6,4,1))
boxplot(log2(counts(dds,normalized=TRUE)+1), ylim=c(-5,20),las=2,cex.names=1.5,main="Size Factor Normalized read counts", ylab="log2(read counts)")

dds = estimateDispersions(dds)
plotDispEsts(dds)
dds = nbinomWaldTest(dds)
res_KPTT_KP = results(dds,contrast=c("Type","KPTT","KP"))
mcols(res_KPTT_KP,use.names=TRUE)

plotMA(res_KPTT_KP)
hist(res_KPTT_KP$pvalue,breaks=20,col="grey")
vsd <- varianceStabilizingTransformation(dds, blind=TRUE)
p = plotPCA(vsd,intgroup="Type")
p

library(tidyverse)
library(ggplot2)
p = plotPCA(vsd,intgroup="Type",returnData=TRUE)
percentVar <- round(100 * attr(p, "percentVar"))
ggplot(p, aes(PC1, PC2, color=Type)) +
  geom_point(size=4) +
  xlab(paste0("PC1: ",percentVar[1],"% variance")) +
  ylab(paste0("PC2: ",percentVar[2],"% variance")) + 
  coord_fixed() +
  xlim(-50,50)+
  ylim(-50,50)+
  theme(aspect.ratio=1)+
  theme(text = element_text(size = 20)) 

sampleDists <- dist(t(assay(vsd)))
library(pheatmap)
library(RColorBrewer)
sampleDistMatrix <- as.matrix(sampleDists)
rownames(sampleDistMatrix) <- paste(vsd$Samples, sep="-")
colnames(sampleDistMatrix) <- NULL
colors <- colorRampPalette( rev(brewer.pal(9, "Blues")) )(255)
pheatmap(sampleDistMatrix,
         clustering_distance_rows=sampleDists,
         clustering_distance_cols=sampleDists,
         col=colors)

res_KPTT_KP = res_KPTT_KP[order(res_KPTT_KP$padj),]
